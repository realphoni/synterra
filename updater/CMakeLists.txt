cmake_minimum_required(VERSION 3.21)
project(synterra_updates VERSION 2.0.2020 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Test)
find_package(LibArchive REQUIRED)
add_library(update-core STATIC release.cpp)
target_link_libraries(update-core PUBLIC Qt6::Network LibArchive::LibArchive)
add_executable(synterra-updates main.cpp)
target_link_libraries(synterra-updates PRIVATE update-core Qt6::Widgets)
add_executable(synterra-release-update apply.cpp)
target_link_libraries(synterra-release-update PRIVATE update-core)
add_executable(synterra-update-manifest manifest.cpp)
target_link_libraries(synterra-update-manifest PRIVATE update-core)
enable_testing()
add_executable(updater-tests tests.cpp)
target_link_libraries(updater-tests PRIVATE update-core Qt6::Test)
add_test(NAME updater COMMAND updater-tests)
install(TARGETS synterra-updates RUNTIME DESTINATION bin)
install(TARGETS synterra-release-update RUNTIME DESTINATION libexec)
