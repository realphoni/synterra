cmake_minimum_required(VERSION 3.21)
project(synaqua VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
set(KDE_INSTALL_PLUGINDIR "lib/qt6/plugins")
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Quick Qml DBus Test)
find_package(KF6Service REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6GlobalAccel REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(LayerShellQt REQUIRED)
add_library(synaqua-settings STATIC docksettings.cpp docksettings.h dockconfig.h)
set_target_properties(synaqua-settings PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(synaqua-settings PUBLIC Qt6::Widgets Qt6::DBus)
kcoreaddons_add_plugin(kcm_synaqua_dock SOURCES dockkcm.cpp INSTALL_NAMESPACE "plasma/kcms/systemsettings")
set_target_properties(kcm_synaqua_dock PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(kcm_synaqua_dock PRIVATE synaqua-settings KF6::KCMUtils KF6::CoreAddons)
qt_add_executable(synaqua-settings-test tests/settings.cpp)
target_link_libraries(synaqua-settings-test PRIVATE synaqua-settings Qt6::Test KF6::KCMUtils KF6::CoreAddons)
qt_add_executable(synaqua-shell main.cpp shell.cpp shell.h)
qt_add_resources(synaqua-shell shell_resources PREFIX /synaqua FILES
    qml/Main.qml qml/DockButton.qml qml/GlassPane.qml
    ../profile/airootfs/usr/share/icons/hicolor/scalable/apps/synterra.svg)
target_link_libraries(synaqua-shell PRIVATE Qt6::Widgets Qt6::Quick Qt6::Qml Qt6::DBus Qt6::Test
    KF6::Service KF6::KIOGui KF6::GlobalAccel KF6::WindowSystem LayerShellQt::Interface synaqua-settings)
target_compile_definitions(synaqua-shell PRIVATE SYNAQUA_VERSION="${PROJECT_VERSION}")
install(TARGETS synaqua-shell RUNTIME DESTINATION bin)
