cmake_minimum_required(VERSION 3.21)
project(synterra_hub VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Test Quick Qml)
set(CMAKE_AUTOMOC ON)
add_library(SynterraGlass MODULE style/synterraglass.cpp)
target_link_libraries(SynterraGlass PRIVATE Qt6::Widgets)
set_target_properties(SynterraGlass PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/styles")
qt_add_executable(synterra-hub main.cpp)
qt_add_resources(synterra-hub hub_resources PREFIX /hub
    BASE ../profile/airootfs/usr/share/icons/hicolor/scalable/apps
    FILES ../profile/airootfs/usr/share/icons/hicolor/scalable/apps/synterra.svg)
target_link_libraries(synterra-hub PRIVATE Qt6::Widgets Qt6::Test)
install(TARGETS synterra-hub RUNTIME DESTINATION bin)
add_executable(synterra-theme-preview tests/theme-preview.cpp)
target_link_libraries(synterra-theme-preview PRIVATE Qt6::Widgets Qt6::Test)
add_executable(synterra-greeter-preview tests/greeter-preview.cpp)
target_link_libraries(synterra-greeter-preview PRIVATE Qt6::Quick Qt6::Qml Qt6::Test)
