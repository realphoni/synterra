cmake_minimum_required(VERSION 3.21)
project(synterra_surf VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebEngineWidgets Test Network)
qt_add_executable(synterra-surf main.cpp notices.cpp)
target_compile_definitions(synterra-surf PRIVATE SURF_VERSION="${PROJECT_VERSION}")
qt_add_resources(synterra-surf surf_resources PREFIX /surf FILES start.html notices.json tests/web-fixture.html)
target_link_libraries(synterra-surf PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::Test Qt6::Network)
install(TARGETS synterra-surf RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    set(CMAKE_AUTOMOC ON)
    qt_add_executable(surf-notice-tests tests/notices-test.cpp notices.cpp)
    qt_add_resources(surf-notice-tests notice_resources PREFIX /surf FILES notices.json)
    target_link_libraries(surf-notice-tests PRIVATE Qt6::Test)
    add_test(NAME surf-notices COMMAND surf-notice-tests)
endif()
